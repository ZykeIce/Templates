//find kth element, with xor apply to the whole array
struct PersistentTrie{
    struct Node{
        int cnt = 0, exist = 0;
        int c[2] = {0, 0};
    };
    int B;
    vector<Node> T;
    vector<int> ver;
    PersistentTrie(int B) : B(B){
        T.push_back(Node());
    };
    void add(int &node, int ori, int x){
        auto add = [&](auto add, int &node, int ori, int i, int x) -> void{
            node = T.size();
            T.push_back(T[ori]);
            T[node].cnt++;
            if(i == 0){
                T[node].exist = 1;
                return;
            }
            Node goat = T[node];
            if(!(1&(x>>(i-1)))){
                add(add, goat.c[0], T[ori].c[0], i-1, x);
            }else{
                add(add, goat.c[1], T[ori].c[1], i-1, x);
            }
            T[node] = goat;
        };
        add(add, node, ori, B+1, x);
    }
    int walk(int node, int ori, int k, int x){
        auto walk = [&](auto walk, int node, int ori, int k, int i, int x) -> int{
            if(T[node].cnt-T[ori].cnt < k) return -1;
            if(i == 0) return 0;
            int lt = (1&(x>>(i-1))), rt = !lt;
            int left = T[T[node].c[lt]].cnt-T[T[ori].c[lt]].cnt;
            if(left >= k){
                return (1<<(i-1))*0 + walk(walk, T[node].c[lt], T[ori].c[lt], k, i-1, x);
            }
            return (1<<(i-1))*1 + walk(walk, T[node].c[rt], T[ori].c[rt], k-left, i-1, x);
        };
        return walk(walk, node, ori, k, B+1, x);
    }
};
