struct PersistentSegmentTree{
    struct Node{
        ll sum = 0, s = -1;
        int ln = 0, rn = 0;
    };
    int n, N;
    vector<Node> nodes;
    vector<int> ver;
    PersistentSegmentTree(int n): n(n){
        N = n;
        nodes.push_back(Node());
        ver.push_back(0);
    };
    void upd(int &node, int ori, int i, int x){
         auto upd = [&](auto upd, int &node, int ori, int low, int high, int i, int x) -> void{
            node = nodes.size();
            nodes.push_back(nodes[ori]);
            if(low == high){
                nodes[node].s = x;
                return;
            }
            int mid = (low+high)/2;
            Node goat = nodes[node];
            if(i <= mid){
                upd(upd, goat.ln, nodes[ori].ln, low, mid, i, x);
            }else{
                upd(upd, goat.rn, nodes[ori].rn, mid+1, high, i, x);
            }
            goat.sum = nodes[goat.ln].sum+nodes[goat.rn].sum;
            nodes[node] = goat;
        };
        upd(upd, node, ori, 1, N, i, x);
    }
    ll get(int node, int l, int r){
        auto get = [&](auto get, int node, int low, int high, int l, int r) -> ll{
            if(low > r || l > high) return 0;
            if(low >= l && r >= high) return nodes[node].sum;
            int mid = (low+high)/2;
            return get(get, nodes[node].ln, low, mid, l, r) + get(get, nodes[node].rn, mid+1, high, l, r);
        };
        return get(get, node, 1, N, l, r);
    }
};
