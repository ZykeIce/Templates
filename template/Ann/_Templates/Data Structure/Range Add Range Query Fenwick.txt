template <class T>
struct FenwickH {
    int n;
    vector<T> bit1, bit2;
    FenwickH(){}
    void init(int x){
        n = x;
        bit1.resize(n+1, 0);
        bit2.resize(n+1, 0);
    }
    void add(int l, int r, T delta){
        for(int i = l; i <= n; i += i & -i) {
            bit1[i] += delta;
        }
        for(int i = l; i <= n; i += i & -i) {
            bit2[i] -= delta * l;
        }
        for(int i = r+1; i <= n; i+=i&-i){
            bit1[i] -= delta;
        }
        for(int i = r+1; i <= n; i+=i&-i){
            bit2[i] -= -delta*(r+1);
        }
    }
    T sum(int l, int r){
        T res = 0;
        for(int i = r; i > 0; i-=i&-i){
            res += bit1[i] * (r+1);
        }
        for(int i = r; i > 0; i-=i&-i){
            res += bit2[i];
        }
        for(int i = l-1; i > 0; i-=i&-i){
            res -= bit1[i] * l;
        }
        for(int i = l-1; i > 0; i-=i&-i){
            res -= bit2[i];
        }
        return res;
    }
};
